# 社内ホストPC（コール履歴を毎日読んでいる PC）の PowerShell に、この下を丸ごと貼って実行する。
# FileMaker の「セキュリティ（アカウント・権限）」が ODBC から見えるかを調べる（読むだけ・書き込みなし）。
# 見るもの：ファイルごとの表の一覧・FileMaker の仕組みの表（FileMaker_Tables／FileMaker_Fields）・名前に「アカウント／権限」を含む表。
# 結果は H:\マイドライブ\東海林専用フォルダ\03_スクリプト・実行結果\probe-security-result.txt にも保存される（フォルダが見えないときは画面だけ）。2026-10-24 作成（入社の流れの自動化：FileMaker のアカウント追加を自動にできるかの前調査）。

$p = "C:\Garden\callcenter-agent\probe-security.ps1"
@'
$ErrorActionPreference = "Continue"
$pw = [Environment]::GetEnvironmentVariable("FM_CALL_HISTORY_PASSWORD","Machine")
if (-not $pw) { $pw = $env:FM_CALL_HISTORY_PASSWORD }
if (-not $pw) { Write-Host "FMパスワードが環境変数に見つかりません"; exit 1 }

# 調べるファイル（ホストされているファイル名。開けないものは「開けない」と出るだけ）
$files = @("コール履歴","【新】営業","新営業","社員情報")
foreach ($db in $files) {
  Write-Host ""
  Write-Host ("##### ファイル：" + $db)
  $cs = "Driver={FileMaker ODBC};Server=192.168.1.100;Port=2399;Database=" + $db + ";UID=Garden;PWD=" + $pw
  $c = New-Object System.Data.Odbc.OdbcConnection $cs
  try { $c.Open() } catch { Write-Host ("  開けない：" + $_.Exception.Message.Split("`n")[0]); continue }
  Write-Host ("  接続OK ServerVersion=" + $c.ServerVersion)
  $tables = @($c.GetSchema("Tables").Rows | ForEach-Object { [string]$_["TABLE_NAME"] })
  Write-Host ("  表の数：" + $tables.Count)
  $tables | ForEach-Object { Write-Host ("    TABLE: " + $_) }
  $hit = $tables | Where-Object { $_ -match "アカウント|account|権限|privilege|セキュリティ|security|password|パスワード" }
  Write-Host ("  名前に アカウント/権限 を含む表：" + ($(if ($hit) { $hit -join ", " } else { "なし" })))
  foreach ($sys in @("FileMaker_Tables","FileMaker_Fields")) {
    $cmd = $c.CreateCommand(); $cmd.CommandTimeout = 60
    $cmd.CommandText = 'SELECT * FROM ' + $sys
    try {
      $r = $cmd.ExecuteReader(); $n = 0
      $cols = @(); for ($i = 0; $i -lt $r.FieldCount; $i++) { $cols += $r.GetName($i) }
      while ($r.Read() -and $n -lt 5) { $n++ }
      $r.Close()
      Write-Host ("  " + $sys + "：読めた（列＝" + ($cols -join ", ") + "）")
    } catch { Write-Host ("  " + $sys + "：読めない（" + $_.Exception.Message.Split("`n")[0] + "）") }
  }
  $c.Close()
}
Write-Host ""
Write-Host "おわり"
'@ | Out-File -FilePath $p -Encoding utf8
$out = "H:\マイドライブ\東海林専用フォルダ\03_スクリプト・実行結果\probe-security-result.txt"
if (Test-Path (Split-Path $out)) {
  & "$env:WINDIR\SysWOW64\WindowsPowerShell\v1.0\powershell.exe" -NoProfile -ExecutionPolicy Bypass -File $p *>&1 | Tee-Object -FilePath $out
  Write-Host ("保存しました：" + $out)
} else {
  & "$env:WINDIR\SysWOW64\WindowsPowerShell\v1.0\powershell.exe" -NoProfile -ExecutionPolicy Bypass -File $p
}
